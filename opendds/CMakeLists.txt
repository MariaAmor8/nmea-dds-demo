cmake_minimum_required(VERSION 3.20)
project(marine_opendds VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenDDS 3.34.0 EXACT REQUIRED OpenDDS::Dcps OpenDDS::Rtps OpenDDS::Rtps_Udp)
include(CTest)

add_library(navigation_idl)
# The shared demo IDL uses the reserved IDL keyword `sequence` as a member.
# Escape it only in the compiler input; IDL's _sequence maps to C++ sequence.
# Scope the macro to this file, so DDS support IDLs keep their sequence<T> syntax.
set(shared_idl "${CMAKE_CURRENT_SOURCE_DIR}/../idl/Navigation.idl")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${shared_idl}")
file(READ "${shared_idl}" navigation_contract)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/idl")
configure_file(compiler_input.idl.in "${CMAKE_CURRENT_BINARY_DIR}/idl/Navigation.idl" @ONLY)
opendds_target_sources(navigation_idl PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/idl/Navigation.idl")
target_link_libraries(navigation_idl PUBLIC OpenDDS::Dcps)

add_library(marine_sources src/sources.cpp src/config.cpp src/udp.cpp)
target_include_directories(marine_sources PUBLIC src)
if(WIN32)
  target_link_libraries(marine_sources PUBLIC ws2_32)
  target_compile_definitions(marine_sources PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(marine_dds src/dds.cpp)
target_link_libraries(marine_dds PUBLIC marine_sources navigation_idl OpenDDS::Rtps OpenDDS::Rtps_Udp)
foreach(role publisher subscriber)
  add_executable(${role} src/${role}.cpp)
  target_link_libraries(${role} PRIVATE marine_dds)
endforeach()
if(MSVC)
  target_compile_options(marine_sources PRIVATE /W4)
  target_compile_options(marine_dds PRIVATE /W4)
else()
  target_compile_options(marine_sources PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(marine_dds PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(source_tests tests/source_tests.cpp)
  target_link_libraries(source_tests PRIVATE marine_sources)
  add_test(NAME sources COMMAND source_tests)
  add_executable(cdr_tests tests/cdr_tests.cpp)
  target_link_libraries(cdr_tests PRIVATE marine_dds)
  add_test(NAME cdr COMMAND cdr_tests)
  option(MARINE_INTEGRATION_TESTS "Run local UDP integration tests (requires working sockets)" OFF)
  if(MARINE_INTEGRATION_TESTS)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME multicast COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" --bin-dir "$<TARGET_FILE_DIR:publisher>")
    set_tests_properties(multicast PROPERTIES TIMEOUT 120)
  endif()
endif()
